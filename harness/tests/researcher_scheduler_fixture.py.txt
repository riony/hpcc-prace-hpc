# Translate nested observer receipts as well as action parameters.
def translate(value):
    if isinstance(value,str) and value.startswith(virtual): return str(physical)+value[len(virtual):]
    if isinstance(value,list): return [translate(v) for v in value]
    if isinstance(value,dict): return {k:translate(v) for k,v in value.items()}
    return value

# Proof digests are over the actual submitted argv. Return virtual paths in the
# hash to match the evidence manifest, exactly as the real SSH transport does.

def virtual_argv(p):

def proof_fields(p):
